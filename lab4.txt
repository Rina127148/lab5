// ---------- Работа с Cookies ----------
function getCookieData($key, $default = []) {
    return isset($_COOKIE[$key]) ? json_decode($_COOKIE[$key], true) : $default;
}

function setCookieData($key, $data, $expire = 0) {
    setcookie($key, json_encode($data), $expire, "/");
}

// Удаляем cookies ошибок после отображения
if ($_SERVER['REQUEST_METHOD'] === 'GET' && isset($_GET['show'])) {
    if (isset($_COOKIE['form_errors'])) setcookie('form_errors', '', time() - 3600, "/");
}

// Получение данных из Cookies
$form_data = getCookieData('form_data');
$form_errors = getCookieData('form_errors');
$success_msg = $_GET['success'] ?? '';

// ---------- Обработка POST (валидация + сохранение) ----------
if ($_SERVER['REQUEST_METHOD'] === 'POST') {
    $data = [
        'full_name' => trim($_POST['full_name'] ?? ''),
        'phone' => trim($_POST['phone'] ?? ''),
        'email' => trim($_POST['email'] ?? ''),
        'birth_date' => $_POST['birth_date'] ?? '',
        'gender' => $_POST['gender'] ?? '',
        'languages' => $_POST['languages'] ?? [],
        'bio' => trim($_POST['bio'] ?? ''),
        'contract' => isset($_POST['contract']) ? 1 : 0
    ];

    $errors = [];

    // Валидация регулярками
    if (empty($data['full_name'])) $errors['full_name'] = "ФИО обязательно";
    elseif (strlen($data['full_name']) > 150) $errors['full_name'] = "ФИО ≤ 150 символов";
    elseif (!preg_match("/^[a-zA-Zа-яА-ЯёЁ\s\-]+$/u", $data['full_name'])) $errors['full_name'] = "Только буквы, пробелы, дефис";

    if (empty($data['phone'])) $errors['phone'] = "Телефон обязателен";
    elseif (!preg_match("/^[\d\s\+\(\)\-]{10,20}$/", $data['phone'])) $errors['phone'] = "Некорректный формат телефона";

    if (empty($data['email'])) $errors['email'] = "Email обязателен";
    elseif (!filter_var($data['email'], FILTER_VALIDATE_EMAIL)) $errors['email'] = "Некорректный email";

    if (empty($data['birth_date'])) $errors['birth_date'] = "Дата рождения обязательна";

    if (!in_array($data['gender'], ['male', 'female', 'other'])) $errors['gender'] = "Выберите пол";

    if (empty($data['languages'])) $errors['languages'] = "Выберите хотя бы один язык";

    if (strlen($data['bio']) > 500) $errors['bio'] = "Биография ≤ 500 символов";

    if (!$data['contract']) $errors['contract'] = "Необходимо принять контракт";

    // Если ошибки -> сохраняем в Cookies + редирект GET
    if (!empty($errors)) {
        setCookieData('form_errors', $errors, time() + 3600);
        setCookieData('form_data', $data, time() + 3600);
        header("Location: index.php");
        exit;
    }

    // ------ Сохранение в БД (успех) ------
    try {
        $stmt = $pdo->prepare("INSERT INTO application SET full_name = ?, phone = ?, email = ?, birth_date = ?, gender = ?, bio = ?, contract_accepted = ?");
        $stmt->execute([$data['full_name'], $data['phone'], $data['email'], $data['birth_date'], $data['gender'], $data['bio'], $data['contract']]);
        $app_id = $pdo->lastInsertId();

        $langStmt = $pdo->prepare("INSERT INTO application_language (application_id, language_id) VALUES (?, ?)");
        foreach ($data['languages'] as $lid) {
            $langStmt->execute([$app_id, $lid]);
        }

        // Успех: сохраняем данные в Cookies на 1 год
        setCookieData('form_data', $data, time() + 365 * 86400);
        setCookieData('form_errors', [], time() - 3600);
        header("Location: index.php?success=1");
        exit;
    } catch (PDOException $e) {
        $errors['db'] = "Ошибка сохранения: " . $e->getMessage();
        setCookieData('form_errors', $errors, time() + 3600);
        setCookieData('form_data', $data, time() + 3600);
        header("Location: index.php");
        exit;
    }
}
